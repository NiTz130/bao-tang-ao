// Step 8: wires inspect mode (light, blur volume, ExhibitInspector, director refs). Safe to run again. Saves the scene.
var cam = Camera.main;
var fps = UnityEngine.Object.FindFirstObjectByType<Museum.FirstPersonController>();
var targeting = cam != null ? cam.GetComponent<Museum.ExhibitTargeting>() : null;
var director = UnityEngine.Object.FindFirstObjectByType<Museum.NarrativeDirector>();
var actions = UnityEditor.AssetDatabase.LoadAssetAtPath<UnityEngine.InputSystem.InputActionAsset>("Assets/InputSystem_Actions.inputactions");
if (cam == null || fps == null || targeting == null || director == null || actions == null)
    return "RESULT_FAIL cam=" + cam + " player=" + fps + " targeting=" + targeting + " director=" + director + " actions=" + actions;

// 1. Light that follows the camera, off until inspect mode.
var lightT = cam.transform.Find("InspectLight");
if (lightT == null)
{
    var g = new GameObject("InspectLight");
    UnityEditor.Undo.RegisterCreatedObjectUndo(g, "InspectLight");
    g.transform.SetParent(cam.transform, false);
    lightT = g.transform;
}
lightT.localPosition = new Vector3(0.35f, 0.3f, 0.3f);
lightT.localRotation = Quaternion.identity;
var light = lightT.GetComponent<Light>();
if (light == null) light = UnityEditor.Undo.AddComponent<Light>(lightT.gameObject);
light.type = LightType.Point;
light.lightmapBakeType = LightmapBakeType.Realtime;
light.shadows = LightShadows.None;
light.range = 2f;
light.color = Color.white;
light.useColorTemperature = true;
light.colorTemperature = 3800f;
light.intensity = 0f;
light.enabled = false;

// 2. Blur profile: depth of field + vignette, no darkening (D3).
const string profilePath = "Assets/Museum/Settings/Museum_InspectProfile.asset";
var profile = UnityEditor.AssetDatabase.LoadAssetAtPath<UnityEngine.Rendering.VolumeProfile>(profilePath);
if (profile == null)
{
    profile = ScriptableObject.CreateInstance<UnityEngine.Rendering.VolumeProfile>();
    UnityEditor.AssetDatabase.CreateAsset(profile, profilePath);
}
UnityEngine.Rendering.Universal.DepthOfField dof;
if (!profile.TryGet(out dof))
{
    dof = profile.Add<UnityEngine.Rendering.Universal.DepthOfField>(true);
    dof.name = "DepthOfField";
    dof.hideFlags = HideFlags.HideInInspector | HideFlags.HideInHierarchy;
    UnityEditor.AssetDatabase.AddObjectToAsset(dof, profile);
}
dof.mode.Override(UnityEngine.Rendering.Universal.DepthOfFieldMode.Gaussian);
dof.gaussianStart.Override(1.2f);
dof.gaussianEnd.Override(4f);
dof.gaussianMaxRadius.Override(1.5f);
dof.highQualitySampling.Override(true);
UnityEngine.Rendering.Universal.Vignette vig;
if (!profile.TryGet(out vig))
{
    vig = profile.Add<UnityEngine.Rendering.Universal.Vignette>(true);
    vig.name = "Vignette";
    vig.hideFlags = HideFlags.HideInInspector | HideFlags.HideInHierarchy;
    UnityEditor.AssetDatabase.AddObjectToAsset(vig, profile);
}
vig.intensity.Override(0.45f);
vig.smoothness.Override(0.4f);
UnityEditor.EditorUtility.SetDirty(dof);
UnityEditor.EditorUtility.SetDirty(vig);
UnityEditor.EditorUtility.SetDirty(profile);
UnityEditor.AssetDatabase.SaveAssets();

// 3. Global volume, weight 0 until inspect mode.
var volGo = GameObject.Find("InspectVolume");
if (volGo == null)
{
    volGo = new GameObject("InspectVolume");
    UnityEditor.Undo.RegisterCreatedObjectUndo(volGo, "InspectVolume");
}
var volume = volGo.GetComponent<UnityEngine.Rendering.Volume>();
if (volume == null) volume = UnityEditor.Undo.AddComponent<UnityEngine.Rendering.Volume>(volGo);
volume.isGlobal = true;
volume.priority = 10f;
volume.weight = 0f;
volume.sharedProfile = profile;

// 4. ExhibitInspector on the Player.
var inspector = fps.GetComponent<Museum.ExhibitInspector>();
if (inspector == null) inspector = UnityEditor.Undo.AddComponent<Museum.ExhibitInspector>(fps.gameObject);
var so = new UnityEditor.SerializedObject(inspector);
so.FindProperty("cam").objectReferenceValue = cam;
so.FindProperty("player").objectReferenceValue = fps;
so.FindProperty("targeting").objectReferenceValue = targeting;
so.FindProperty("inspectLight").objectReferenceValue = light;
so.FindProperty("inspectVolume").objectReferenceValue = volume;
so.FindProperty("actions").objectReferenceValue = actions;
so.ApplyModifiedProperties();

// 5. Director refs.
var ds = new UnityEditor.SerializedObject(director);
ds.FindProperty("targeting").objectReferenceValue = targeting;
ds.FindProperty("inspector").objectReferenceValue = inspector;
ds.ApplyModifiedProperties();

var scene = UnityEngine.SceneManagement.SceneManager.GetActiveScene();
UnityEditor.SceneManagement.EditorSceneManager.MarkSceneDirty(scene);
UnityEditor.SceneManagement.EditorSceneManager.SaveScene(scene);

so.Update(); ds.Update();
int n = UnityEngine.Object.FindObjectsByType<Museum.ExhibitInspector>(FindObjectsSortMode.None).Length;
bool wired = true;
foreach (var f in new[] { "cam", "player", "targeting", "inspectLight", "inspectVolume", "actions" })
    if (so.FindProperty(f).objectReferenceValue == null) wired = false;
bool directorWired = ds.FindProperty("targeting").objectReferenceValue != null && ds.FindProperty("inspector").objectReferenceValue != null;
bool hasDof = profile.Has<UnityEngine.Rendering.Universal.DepthOfField>();
bool ok = n == 1 && wired && directorWired && hasDof && volume.weight == 0f && !light.enabled;
return (ok ? "RESULT_OK" : "RESULT_FAIL") + " inspector=" + n + " light=" + light.name + " volume=" + volGo.name + " prio=" + volume.priority + " dof=" + hasDof + " director=" + (directorWired ? "wired" : "MISSING") + " dirty=" + scene.isDirty;

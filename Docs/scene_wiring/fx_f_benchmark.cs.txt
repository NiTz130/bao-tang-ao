// Step 9: wires PerfRecorder on the Player and 5 benchmark points under the root "Benchmark". Safe to run again. Saves the scene.
// Points stand at the player's spawn height, one per zone, and face the exhibits of that zone. Coordinates: Docs/plans/2026-10-16-rotunda-rebuild.md.
var fps = UnityEngine.Object.FindFirstObjectByType<Museum.FirstPersonController>();
if (fps == null) return "RESULT_FAIL no Player";
var player = fps.transform;
float y = player.position.y;

// name, x, z, yaw (0 = +z north, 90 = +x east, 270 = -x west)
var spec = new object[,]
{
    { "Bench_Vestibule", 0f,    -13.5f, 0f },   // toward the Dong Son drum at (0, -11)
    { "Bench_Rotunda",   0f,    -6f,    0f },   // toward the Tu Duc stele at (0, 0)
    { "Bench_West",      -10.5f, 0f,    270f }, // behind the arch, toward Exhibit_01 and 02
    { "Bench_North",     0f,    10.5f,  0f },   // toward Exhibit_03 and 04
    { "Bench_East",      10.5f, 0f,     90f },  // toward Exhibit_05
};

var root = GameObject.Find("Benchmark");
if (root == null)
{
    root = new GameObject("Benchmark");
    UnityEditor.Undo.RegisterCreatedObjectUndo(root, "Benchmark");
}
root.isStatic = false;
root.transform.SetPositionAndRotation(Vector3.zero, Quaternion.identity);

var points = new Transform[spec.GetLength(0)];
for (int i = 0; i < points.Length; i++)
{
    var name = (string)spec[i, 0];
    var t = root.transform.Find(name);
    if (t == null)
    {
        var g = new GameObject(name);
        UnityEditor.Undo.RegisterCreatedObjectUndo(g, name);
        g.transform.SetParent(root.transform, false);
        t = g.transform;
    }
    t.gameObject.isStatic = false;
    t.SetPositionAndRotation(new Vector3((float)spec[i, 1], y, (float)spec[i, 2]), Quaternion.Euler(0f, (float)spec[i, 3], 0f));
    points[i] = t;
}

var rec = player.GetComponent<Museum.PerfRecorder>();
if (rec == null) rec = UnityEditor.Undo.AddComponent<Museum.PerfRecorder>(player.gameObject);
var so = new UnityEditor.SerializedObject(rec);
so.FindProperty("player").objectReferenceValue = player;
var arr = so.FindProperty("benchmarkPoints");
arr.arraySize = points.Length;
for (int i = 0; i < points.Length; i++) arr.GetArrayElementAtIndex(i).objectReferenceValue = points[i];
so.FindProperty("settleSeconds").floatValue = 2f;
so.FindProperty("recordSeconds").floatValue = 10f;
so.ApplyModifiedProperties();

var scene = UnityEngine.SceneManagement.SceneManager.GetActiveScene();
UnityEditor.SceneManagement.EditorSceneManager.MarkSceneDirty(scene);
UnityEditor.SceneManagement.EditorSceneManager.SaveScene(scene);

so.Update();
int nRec = UnityEngine.Object.FindObjectsByType<Museum.PerfRecorder>(FindObjectsSortMode.None).Length;
int nRoot = 0;
foreach (var g in scene.GetRootGameObjects()) if (g.name == "Benchmark") nRoot++;
bool noAnchor = root.GetComponentInChildren<Museum.NarrativeAnchor>(true) == null;
bool ok = nRec == 1 && nRoot == 1 && root.transform.childCount == 5 && arr.arraySize == 5 && noAnchor && !root.isStatic
    && so.FindProperty("player").objectReferenceValue == player;
var sb = new System.Text.StringBuilder();
foreach (var p in points) sb.Append(p.name + "=" + p.position + "/yaw" + p.eulerAngles.y + " ");
return (ok ? "RESULT_OK" : "RESULT_FAIL") + " recorders=" + nRec + " roots=" + nRoot + " children=" + root.transform.childCount + " " + sb + "dirty=" + scene.isDirty;
